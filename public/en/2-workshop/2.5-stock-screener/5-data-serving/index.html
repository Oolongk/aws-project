Architectural Thinking – Access Pattern First. Using ticker + snapshot_date enables the Reader Lambda to use a Query operation to retrieve the historical series of a specific symbol. Avoid using Scan to populate charts.

qmj_transform.py writes Parquet files to processed_qmj/as_of_date=YYYY-MM-DD/. This section walks you through deploying the Lambda QMJ Loader to receive S3 ObjectCreated events, read the Parquet files using awswrangler, and perform a batch write to DynamoDB.

This Lambda functions in two modes:

Mode Trigger / Invocation Purpose
s3_trigger S3 automatically sends an event when a new .parquet is created Incremental loading of new outputs from Glue PySpark
backfill Lambda Test Event or manual invocation API Scans the entire processed_qmj/ prefix and loads historical QMJ data

Architectural Thinking – Source Boundary. This is not a Glue Publisher Job. S3 acts as the event source, Lambda serves as the compute layer loading the Serving Store, and DynamoDB acts as the read-optimized projection for the website.

Lambda Loader Source Code

The line import json is included because the source needs json.dumps(response) when failed_files are encountered.

import json
import os
import re
import math
import boto3
import urllib.parse
import awswrangler as wr
import pandas as pd

from datetime import date, datetime
from decimal import Decimal
from numbers import Integral, Real

TABLE_NAME = os.environ.get("TABLE_NAME", "qmj_screener")
PROCESSED_BUCKET = os.environ.get(
    "PROCESSED_BUCKET",
    "stock-screener-processed-your-name"
)
PROCESSED_PREFIX = os.environ.get(
    "PROCESSED_PREFIX",
    "processed_qmj/"
)

dynamodb = boto3.resource("dynamodb")
s3 = boto3.client("s3")
table = dynamodb.Table(TABLE_NAME)


def to_dynamodb_value(value):
    # Standardize Pandas/NumPy values to DynamoDB-compatible types.
    if value is None:
        return None

    try:
        if bool(pd.isna(value)):
            return None
    except Exception:
        pass

    if isinstance(value, (datetime, date)):
        return value.strftime("%Y-%m-%d")

    if isinstance(value, bool):
        return value

    if isinstance(value, Decimal):
        return value

    if isinstance(value, Integral):
        return Decimal(str(int(value)))

    if isinstance(value, Real):
        number = float(value)
        if not math.isfinite(number):
            return None
        return Decimal(str(number))

    if isinstance(value, str):
        return value

    return str(value)


def extract_as_of_date(s3_key):
    # Extract partition date from processed_qmj/as_of_date=YYYY-MM-DD/.
    match = re.search(
        r"(?:^|/)as_of_date=(\d{4}-\d{2}-\d{2})(?:/|$)",
        s3_key
    )

    if not match:
        raise ValueError(f"as_of_date not found in key: {s3_key}")

    as_of_date = match.group(1)
    datetime.strptime(as_of_date, "%Y-%m-%d")
    return as_of_date


def load_parquet_to_dynamodb(bucket, key):
    as_of_date = extract_as_of_date(key)
    s3_path = f"s3://{bucket}/{key}"
    df = wr.s3.read_parquet(path=s3_path)

    if df.empty:
        return 0, 0

    if "ticker" not in df.columns:
        raise ValueError(f"File missing ticker column: {s3_path}")

    loaded_rows = 0
    skipped_rows = 0
    columns = list(df.columns)

    # Overwrite identical ticker + snapshot_date so retry events do not create duplicate logical items.
    with table.batch_writer(
        overwrite_by_pkeys=["ticker", "snapshot_date"]
    ) as batch:
        for values in df.itertuples(index=False, name=None):
            item = {}

            for column, value in zip(columns, values):
                clean_value = to_dynamodb_value(value)
                if clean_value is not None:
                    item[column] = clean_value

            ticker = str(item.get("ticker", "")).strip().upper()
            if not ticker:
                skipped_rows += 1
                continue

            item["ticker"] = ticker
            item["snapshot_date"] = as_of_date
            item["as_of_date"] = as_of_date
            item["source_s3_key"] = key

            batch.put_item(Item=item)
            loaded_rows += 1

    return loaded_rows, skipped_rows


def get_existing_parquet_files(bucket, prefix):
    keys = []
    paginator = s3.get_paginator("list_objects_v2")

    for page in paginator.paginate(Bucket=bucket, Prefix=prefix):
        for obj in page.get("Contents", []):
            key = obj["Key"]
            if key.endswith(".parquet"):
                keys.append(key)

    return sorted(keys)


def process_files(bucket, keys):
    processed_files = 0
    loaded_rows = 0
    skipped_rows = 0
    failed_files = []

    for key in keys:
        try:
            loaded, skipped = load_parquet_to_dynamodb(bucket, key)
            processed_files += 1
            loaded_rows += loaded
            skipped_rows += skipped
        except Exception as error:
            failed_files.append({"key": key, "error": str(error)})

    return {
        "processed_files": processed_files,
        "loaded_rows": loaded_rows,
        "skipped_rows": skipped_rows,
        "failed_files": failed_files
    }


def lambda_handler(event, context):
    event = event or {}

    # Manual backfill mode: scan all QMJ Parquet files in the Processed Zone.
    if event.get("mode") == "backfill":
        bucket = event.get("bucket", PROCESSED_BUCKET)
        prefix = event.get("prefix", PROCESSED_PREFIX)
        keys = get_existing_parquet_files(bucket, prefix)

        if not keys:
            return {
                "mode": "backfill",
                "processed_files": 0,
                "loaded_rows": 0,
                "skipped_rows": 0,
                "failed_files": []
            }

        result = process_files(bucket, keys)
        result["mode"] = "backfill"
        result["total_parquet_files_found"] = len(keys)
        return result

    # S3 trigger mode: filter ObjectCreated events under the Processed QMJ prefix.
    keys_by_bucket = {}

    for record in event.get("Records", []):
        if record.get("eventSource") != "aws:s3":
            continue

        if not record.get("eventName", "").startswith("ObjectCreated:"):
            continue

        bucket = record["s3"]["bucket"]["name"]
        key = urllib.parse.unquote_plus(record["s3"]["object"]["key"])

        if not key.startswith(PROCESSED_PREFIX) or not key.endswith(".parquet"):
            continue

        keys_by_bucket.setdefault(bucket, []).append(key)

    total_processed_files = 0
    total_loaded_rows = 0
    total_skipped_rows = 0
    failed_files = []

    for bucket, keys in keys_by_bucket.items():
        result = process_files(bucket, keys)
        total_processed_files += result["processed_files"]
        total_loaded_rows += result["loaded_rows"]
        total_skipped_rows += result["skipped_rows"]
        failed_files.extend(result["failed_files"])

    response = {
        "mode": "s3_trigger",
        "processed_files": total_processed_files,
        "loaded_rows": total_loaded_rows,
        "skipped_rows": total_skipped_rows,
        "failed_files": failed_files
    }

    if failed_files:
        raise Exception(json.dumps(response))

    return response

import json
import os
from decimal import Decimal

import boto3
from boto3.dynamodb.conditions import Key

TABLE_NAME = os.environ["DYNAMODB_TABLE"]
table = boto3.resource("dynamodb").Table(TABLE_NAME)


def json_default(value):
    # DynamoDB returns Decimal; API response requires a JSON number.
    if isinstance(value, Decimal):
        return float(value)
    raise TypeError(f"Unsupported type: {type(value)}")


def response(status_code, body):
    return {
        "statusCode": status_code,
        "headers": {
            "Content-Type": "application/json",
            "Access-Control-Allow-Origin": "*",
            "Access-Control-Allow-Methods": "GET,OPTIONS",
            "Access-Control-Allow-Headers": "Content-Type"
        },
        "body": json.dumps(body, default=json_default)
    }


def lambda_handler(event, context):
    params = event.get("queryStringParameters") or {}
    ticker = (params.get("ticker") or "").strip().upper()

    if not ticker:
        return response(400, {"message": "ticker is required"})

    result = table.query(
        KeyConditionExpression=Key("ticker").eq(ticker),
        ScanIndexForward=True
    )

    return response(200, {
        "ticker": ticker,
        "items": result.get("Items", [])
    })

Architectural Thinking – Query, Not Scan. The Reader only has dynamodb:Query permissions. It is not granted dynamodb:Scan capability, does not require S3 permissions, and does not use the Loader Role.

Architectural Thinking – Serving Separation. The browser does not load Parquet files from S3 Processed. Parquet files are published to DynamoDB, and the API returns only the JSON fields required for the chart.

This chapter covers bringing the QMJ Parquet files from S3 Processed to the website using an event-driven architecture: S3 triggers the Lambda QMJ Loader, the Loader writes to DynamoDB, the Lambda Reader queries by ticker, API Gateway returns JSON, and the Web Frontend renders charts using Chart.js.

The EventBridge Scheduler triggers the Glue Python Shell Job daily. Do not configure the scheduler to run the transform or publisher tasks at the same time, as they must wait for the Raw output to be fully completed.
