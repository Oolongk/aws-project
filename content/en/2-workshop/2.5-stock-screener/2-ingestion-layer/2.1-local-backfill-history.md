The local backfill script ingests historical price data from `2023-01-01` to the current date. The code filters S&P 500 by `MIN_MARKET_CAP = 10_000_000_000`, partitions prices by individual trading date, and writes a company info snapshot on the day the script runs.

The local backfill script does not write `us_financials_raw.parquet`. Financials are ingested via the daily ingestion pipeline; the transformation code includes appropriate fallback handlers.

````python
import io
import time
from datetime import datetime, timedelta

import boto3
import pandas as pd
import requests
import yfinance as yf
from botocore.exceptions import ClientError
from tqdm import tqdm

# =========================================================
# AWS S3 CONFIG
# =========================================================
BUCKET_NAME = "stock-screener-raw-your-name"
RAW_ROOT = "raw_us_market"

# boto3 retrieves credentials from the AWS CLI profile, environment variables, or IAM Role.
# Do not hardcode Access Key or Secret Key in source code.

# Must match the region of your bucket
AWS_REGION = "ap-southeast-1"

# =========================================================
# BACKFILL CONFIG
# =========================================================
MIN_MARKET_CAP = 10_000_000_000  # 10 billion USD

# Date range for price backfill
BACKFILL_START_DATE = "2023-01-01"
BACKFILL_END_DATE = (datetime.now() + timedelta(days=1)).strftime("%Y-%m-%d")

# Test with 30 tickers first. Set to None to run the full S&P 500
MAX_TICKERS_FOR_TEST = None

# =========================================================
# AWS CLIENTS
# =========================================================
# Use the default credential provider chain instead of embedding credentials in the file.
session = boto3.Session(region_name=AWS_REGION)
s3 = session.client("s3")
sts = session.client("sts")


# =========================================================
# AWS CHECK
# =========================================================
def verify_aws_connection():
    try:
        identity = sts.get_caller_identity()

        print("✅ AWS authentication successful")
        print(f"   Account: {identity['Account']}")
        print(f"   ARN: {identity['Arn']}")

    except ClientError as e:
        raise RuntimeError(
            f"❌ AWS credentials error: {e.response['Error']['Code']}"
        ) from e


# =========================================================
# PARQUET + S3
# =========================================================
def to_parquet_bytes(df: pd.DataFrame) -> bytes:
    """
    Write Parquet compatible with AWS Glue:
    - Avoid datetime64[ns] / TIMESTAMP(NANOS)
    - Date columns are saved as string YYYY-MM-DD
    """
    output_df = df.copy()

    if not isinstance(output_df.index, pd.RangeIndex):
        output_df = output_df.reset_index()

    # Rename Date/Datetime from yfinance to price_date
    if "Date" in output_df.columns:
        output_df = output_df.rename(columns={"Date": "price_date"})

    if "Datetime" in output_df.columns:
        output_df = output_df.rename(columns={"Datetime": "price_date"})

    # Convert all datetime columns to string so Glue does not encounter TIMESTAMP(NANOS)
    for column_name in output_df.columns:
        if pd.api.types.is_datetime64_any_dtype(output_df[column_name]):
            date_series = output_df[column_name]

            # Remove timezone but preserve the trading date
            if isinstance(date_series.dtype, pd.DatetimeTZDtype):
                date_series = date_series.dt.tz_localize(None)

            # price_date / snapshot_date only need date precision
            if column_name in ["price_date", "snapshot_date"]:
                output_df[column_name] = date_series.dt.strftime("%Y-%m-%d")
            else:
                output_df[column_name] = date_series.dt.strftime(
                    "%Y-%m-%d %H:%M:%S"
                )

    buffer = io.BytesIO()

    output_df.to_parquet(
        buffer,
        index=False,
        compression="snappy",
        engine="pyarrow",
        version="1.0"
    )

    return buffer.getvalue()


def put_df(df: pd.DataFrame, key: str):
    """Upload a DataFrame to S3 as Parquet."""
    if df.empty:
        print(f"⚠️ Skipping {key}: empty DataFrame.")
        return

    try:
        s3.put_object(
            Bucket=BUCKET_NAME,
            Key=key,
            Body=to_parquet_bytes(df),
            ContentType="application/octet-stream"
        )

        print(f"✅ Uploaded: s3://{BUCKET_NAME}/{key} | {len(df)} rows")

    except ClientError as e:
        raise RuntimeError(
            f"❌ Cannot upload to S3.\n"
            f"Error: {e.response['Error']['Code']}\n"
            f"Key: {key}"
        ) from e


# =========================================================
# TICKER SOURCE
# =========================================================
def get_sp500_tickers():
    url = "https://en.wikipedia.org/wiki/List_of_S%26P_500_companies"

    headers = {
        "User-Agent": (
            "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
            "AppleWebKit/537.36 Chrome/120.0 Safari/537.36"
        )
    }

    response = requests.get(url, headers=headers, timeout=30)
    response.raise_for_status()

    df_sp500 = pd.read_html(response.text)[0]

    # yfinance uses BRK-B instead of BRK.B
    return [
        ticker.replace(".", "-")
        for ticker in df_sp500["Symbol"].tolist()
    ]

# =========================================================
# COMPANY INFO (SCHEMA ERROR FIX — MISSING COLUMNS)
# =========================================================
def get_company_info(ticker: str, ingestion_date: str):
    """
    Company info from yfinance is current data.
    Schema is 100% synced with Daily Ingestion code so Spark does not report errors.
    """
    stock = yf.Ticker(ticker)

    try:
        fast_info = stock.fast_info or {}
        market_cap = fast_info.get("marketCap", 0)
    except Exception:
        fast_info = {}
        market_cap = 0

    if not market_cap or market_cap < MIN_MARKET_CAP:
        return None

    try:
        info = stock.info or {}
    except Exception:
        info = {}

    return {
        "ticker": ticker,
        "snapshot_date": ingestion_date,
        "company_name": info.get("shortName") or info.get("longName") or ticker,
        "exchange": info.get("exchange"),
        "sector": info.get("sector"),
        "industry": info.get("industry"),
        "country": info.get("country"),
        "currency": info.get("currency"),
        "website": info.get("website"),
        "current_market_cap": market_cap or info.get("marketCap"),
        "shares_outstanding": info.get("sharesOutstanding"),
        "enterprise_value": info.get("enterpriseValue"),
        "float_shares": info.get("floatShares"),
        "beta": info.get("beta"),
        "trailing_pe": info.get("trailingPE"),
        "return_on_equity": info.get("returnOnEquity"),
        "debt_to_equity": info.get("debtToEquity"),
        "earnings_growth": info.get("earningsGrowth"),
        "revenue_growth": info.get("revenueGrowth"),
        "ingested_at": datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    }


# =========================================================
# PRICE HISTORY (SCHEMA ERROR FIX)
# =========================================================
def get_price_history(ticker: str, ingestion_date: str):
    """
    Fetch price history in the range BACKFILL_START_DATE -> BACKFILL_END_DATE.
    Schema is standardized (column names in lowercase) so Spark does not report errors.
    """
    stock = yf.Ticker(ticker)

    df_price = stock.history(
        start=BACKFILL_START_DATE,
        end=BACKFILL_END_DATE,
        auto_adjust=False
    )

    if df_price.empty:
        return pd.DataFrame()

    # Promote the Date index to a price_date column
    df_price = df_price.reset_index()

    # Process date columns
    if "Date" in df_price.columns:
        df_price = df_price.rename(columns={"Date": "price_date"})
    if "Datetime" in df_price.columns:
        df_price = df_price.rename(columns={"Datetime": "price_date"})

    df_price["price_date"] = pd.to_datetime(df_price["price_date"]).dt.tz_localize(None)

    # ---------------------------------------------------------
    # SCHEMA FIX: Rename yfinance column names -> AWS Glue names
    # ---------------------------------------------------------
    rename_map = {
        "Open": "open",
        "High": "high",
        "Low": "low",
        "Close": "close",
        "Adj Close": "adj_close",
        "Volume": "volume",
        "Dividends": "dividends",
        "Stock Splits": "stock_splits"
    }
    
    # Rename existing columns
    df_price = df_price.rename(columns=rename_map)

    # If a newer version of yfinance does not return Adj Close when auto_adjust=False
    if "adj_close" not in df_price.columns and "close" in df_price.columns:
        df_price["adj_close"] = df_price["close"]

    # Cast data types (Cast type) to float64 to avoid Integer vs Double errors in Spark
    numeric_columns = ["open", "high", "low", "close", "adj_close", "volume", "dividends", "stock_splits"]
    for col in numeric_columns:
        if col in df_price.columns:
            df_price[col] = pd.to_numeric(df_price[col], errors="coerce").astype("float64")

    # Add metadata
    df_price["ticker"] = ticker
    df_price["snapshot_date"] = ingestion_date
    df_price["ingested_at"] = datetime.now().strftime("%Y-%m-%d %H:%M:%S")

    # Only keep required columns in the correct order
    expected_cols = [
        "price_date", "ticker", "open", "high", "low", "close", 
        "adj_close", "volume", "dividends", "stock_splits", 
        "snapshot_date", "ingested_at"
    ]
    
    # Filter to only columns that actually exist (avoid KeyError if yfinance is missing columns)
    final_cols = [c for c in expected_cols if c in df_price.columns]
    df_price = df_price[final_cols]

    return df_price


# =========================================================
# BACKFILL PARTITION UPLOAD
# =========================================================
def upload_prices_partitioned_by_date(df_prices: pd.DataFrame):
    """
    Upload by individual trading date.

    Example:
    raw_us_market/2023/01/03/us_prices_raw.parquet
    raw_us_market/2023/01/04/us_prices_raw.parquet
    """
    if df_prices.empty:
        print("⚠️ No price data to partition.")
        return

    df_prices = df_prices.copy()

    df_prices["price_date"] = pd.to_datetime(
        df_prices["price_date"],
        errors="coerce"
    )

    df_prices = df_prices.dropna(subset=["price_date"])

    grouped = df_prices.groupby(
        df_prices["price_date"].dt.normalize(),
        sort=True
    )

    print(f"\n☁️ Uploading {len(grouped)} date partitions...")

    for trade_date, df_one_day in tqdm(
        grouped,
        total=len(grouped),
        desc="Uploading date partitions"
    ):
        yyyy = trade_date.strftime("%Y")
        mm = trade_date.strftime("%m")
        dd = trade_date.strftime("%d")

        key = (
            f"{RAW_ROOT}/"
            f"{yyyy}/{mm}/{dd}/"
            f"us_prices_raw.parquet"
        )

        put_df(df_one_day, key)


def upload_company_info_current_snapshot(
    df_company_info: pd.DataFrame,
    ingestion_datetime: datetime
):
    """
    Company info is only a current snapshot,
    so it is saved by the date the pipeline runs.
    Added float64 casting to prevent INT64 vs DOUBLE errors in Spark.
    """
    if df_company_info.empty:
        return

    output_df = df_company_info.copy()

    # Force cast all metric columns to float64 (Double in Spark)
    numeric_columns = [
        "current_market_cap", "shares_outstanding", "enterprise_value",
        "float_shares", "beta", "trailing_pe", "return_on_equity",
        "debt_to_equity", "earnings_growth", "revenue_growth"
    ]

    for col in numeric_columns:
        if col in output_df.columns:
            output_df[col] = pd.to_numeric(output_df[col], errors="coerce").astype("float64")

    yyyy = ingestion_datetime.strftime("%Y")
    mm = ingestion_datetime.strftime("%m")
    dd = ingestion_datetime.strftime("%d")

    key = (
        f"{RAW_ROOT}/"
        f"{yyyy}/{mm}/{dd}/"
        f"us_company_info.parquet"
    )

    put_df(output_df, key)


# =========================================================
# MAIN PIPELINE
# =========================================================
def main():
    print("=" * 70)
    print("🚀 STOCK SCREENER - BACKFILL RAW INGESTION")
    print("=" * 70)

    verify_aws_connection()

    run_datetime = datetime.now()
    ingestion_date = run_datetime.strftime("%Y-%m-%d")

    print(f"\n📅 Backfill start: {BACKFILL_START_DATE}")
    print(f"📅 Backfill end:   {BACKFILL_END_DATE}")

    # -----------------------------------------------------
    # STEP 1: FETCH TICKERS
    # -----------------------------------------------------
    print("\n🚀 STEP 1: FETCH S&P 500 TICKER LIST")

    all_tickers = get_sp500_tickers()

    if MAX_TICKERS_FOR_TEST is not None:
        all_tickers = all_tickers[:MAX_TICKERS_FOR_TEST]

    print(f"✅ Tickers to process: {len(all_tickers)}")

    # -----------------------------------------------------
    # STEP 2: FILTER BY MARKET CAP
    # -----------------------------------------------------
    print("\n🚀 STEP 2: FILTER TICKERS BY MARKET CAP")

    valid_tickers = []
    all_company_info = []

    for ticker in tqdm(all_tickers, desc="Filtering tickers"):
        try:
            company_info = get_company_info(
                ticker=ticker,
                ingestion_date=ingestion_date
            )

            if company_info is not None:
                valid_tickers.append(ticker)
                all_company_info.append(company_info)

        except Exception as e:
            print(f"\n⚠️ Error fetching company info for {ticker}: {e}")

        time.sleep(0.2)

    print(f"\n🎯 {len(valid_tickers)} tickers passed the filter.")

    # -----------------------------------------------------
    # STEP 3: BACKFILL PRICE HISTORY
    # -----------------------------------------------------
    print("\n🚀 STEP 3: BACKFILL PRICE HISTORY")

    all_prices = []

    for ticker in tqdm(valid_tickers, desc="Fetching price history"):
        try:
            df_price = get_price_history(
                ticker=ticker,
                ingestion_date=ingestion_date
            )

            if not df_price.empty:
                all_prices.append(df_price)

        except Exception as e:
            print(f"\n⚠️ Error fetching prices for {ticker}: {e}")

        time.sleep(0.2)

    # -----------------------------------------------------
    # STEP 4: UPLOAD TO S3
    # -----------------------------------------------------
    print("\n☁️ STEP 4: UPLOAD TO S3")

    if all_prices:
        df_prices = pd.concat(all_prices, ignore_index=True)

        print(f"✅ Total price rows: {len(df_prices):,}")

        upload_prices_partitioned_by_date(df_prices)
    else:
        print("⚠️ No price data to upload.")

    if all_company_info:
        df_company_info = pd.DataFrame(all_company_info)

        upload_company_info_current_snapshot(
            df_company_info=df_company_info,
            ingestion_datetime=run_datetime
        )
    else:
        print("⚠️ No company info to upload.")

    print("\n🎉 BACKFILL COMPLETE!")
    print(f"📁 Bucket: s3://{BUCKET_NAME}/{RAW_ROOT}/")


if __name__ == "__main__":
    main()
````
