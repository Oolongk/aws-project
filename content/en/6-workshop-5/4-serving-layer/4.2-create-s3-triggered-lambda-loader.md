`qmj_transform.py` outputs Parquet files to `processed_qmj/as_of_date=YYYY-MM-DD/`. This section configures Lambda QMJ Loader to ingest newly processed files.

This Lambda function operates in two modes:

| Mode | Invocation Method | Purpose |
|---|---|---|
| `s3_trigger` | S3 automatically sends event notifications when new `.parquet` files arrive | Ingests new incremental output from Glue PySpark |
| `backfill` | Lambda Test Event or Invoke API | Scans the entire `processed_qmj/` directory and loads historical QMJ scores |

> **Architectural Principle – Source Boundary.** This is not a Glue Publisher Job. S3 is the event source, Lambda is the compute layer that loads the Serving Store, and DynamoDB is the read-optimized projection for the website.

# Prepare Deployment Package

The source imports `awswrangler`, `pandas`, and `pyarrow` as dependencies. The default Lambda runtime does not include these packages, so you need to attach a Lambda Layer or upload a deployment package compatible with the Python runtime and architecture you are using.

| Step | Action | Technical Details |
|---|---|---|
| 1 | Open AWS CloudShell or a Lambda-compatible build environment | Used to create a Linux-compatible package |
| 2 | Create a package directory | Package must contain `lambda_function.py` and dependencies |
| 3 | Install dependencies | Install `awswrangler`, `pandas`, `pyarrow` into the package or a Layer |
| 4 | Copy Lambda source | Save the source below as `lambda_function.py` |
| 5 | Zip the package | Upload the zip to Lambda in the next step |

````bash
# Run in a Lambda-compatible build environment.
mkdir qmj-loader-package
cd qmj-loader-package

# Install Lambda source dependencies into the package.
python -m pip install --target . awswrangler pandas pyarrow
cp ../lambda_qmj_loader.py lambda_function.py

# Create the upload file for Lambda.
zip -r ../qmj-loader.zip .
````

# Lambda Loader Source Code

The `import json` line is included because the source needs `json.dumps(response)` when `failed_files` are present.

````python
import json
import os
import re
import math
import boto3
import urllib.parse
import awswrangler as wr
import pandas as pd

from datetime import date, datetime
from decimal import Decimal
from numbers import Integral, Real

TABLE_NAME = os.environ.get("TABLE_NAME", "qmj_screener")
PROCESSED_BUCKET = os.environ.get(
    "PROCESSED_BUCKET",
    "stock-screener-processed-your-name"
)
PROCESSED_PREFIX = os.environ.get(
    "PROCESSED_PREFIX",
    "processed_qmj/"
)

dynamodb = boto3.resource("dynamodb")
s3 = boto3.client("s3")
table = dynamodb.Table(TABLE_NAME)


def to_dynamodb_value(value):
    # Normalize Pandas/NumPy values to types accepted by DynamoDB.
    if value is None:
        return None

    try:
        if bool(pd.isna(value)):
            return None
    except Exception:
        pass

    if isinstance(value, (datetime, date)):
        return value.strftime("%Y-%m-%d")

    if isinstance(value, bool):
        return value

    if isinstance(value, Decimal):
        return value

    if isinstance(value, Integral):
        return Decimal(str(int(value)))

    if isinstance(value, Real):
        number = float(value)
        if not math.isfinite(number):
            return None
        return Decimal(str(number))

    if isinstance(value, str):
        return value

    return str(value)


def extract_as_of_date(s3_key):
    # Extract partition date from processed_qmj/as_of_date=YYYY-MM-DD/.
    match = re.search(
        r"(?:^|/)as_of_date=(\d{4}-\d{2}-\d{2})(?:/|$)",
        s3_key
    )

    if not match:
        raise ValueError(f"as_of_date not found in key: {s3_key}")

    as_of_date = match.group(1)
    datetime.strptime(as_of_date, "%Y-%m-%d")
    return as_of_date


def load_parquet_to_dynamodb(bucket, key):
    as_of_date = extract_as_of_date(key)
    s3_path = f"s3://{bucket}/{key}"
    df = wr.s3.read_parquet(path=s3_path)

    if df.empty:
        return 0, 0

    if "ticker" not in df.columns:
        raise ValueError(f"File does not have ticker column: {s3_path}")

    loaded_rows = 0
    skipped_rows = 0
    columns = list(df.columns)

    # Overwrite by ticker + snapshot_date so retried events do not create duplicate logical items.
    with table.batch_writer(
        overwrite_by_pkeys=["ticker", "snapshot_date"]
    ) as batch:
        for values in df.itertuples(index=False, name=None):
            item = {}

            for column, value in zip(columns, values):
                clean_value = to_dynamodb_value(value)
                if clean_value is not None:
                    item[column] = clean_value

            ticker = str(item.get("ticker", "")).strip().upper()
            if not ticker:
                skipped_rows += 1
                continue

            item["ticker"] = ticker
            item["snapshot_date"] = as_of_date
            item["as_of_date"] = as_of_date
            item["source_s3_key"] = key

            batch.put_item(Item=item)
            loaded_rows += 1

    return loaded_rows, skipped_rows


def get_existing_parquet_files(bucket, prefix):
    keys = []
    paginator = s3.get_paginator("list_objects_v2")

    for page in paginator.paginate(Bucket=bucket, Prefix=prefix):
        for obj in page.get("Contents", []):
            key = obj["Key"]
            if key.endswith(".parquet"):
                keys.append(key)

    return sorted(keys)


def process_files(bucket, keys):
    processed_files = 0
    loaded_rows = 0
    skipped_rows = 0
    failed_files = []

    for key in keys:
        try:
            loaded, skipped = load_parquet_to_dynamodb(bucket, key)
            processed_files += 1
            loaded_rows += loaded
            skipped_rows += skipped
        except Exception as error:
            failed_files.append({"key": key, "error": str(error)})

    return {
        "processed_files": processed_files,
        "loaded_rows": loaded_rows,
        "skipped_rows": skipped_rows,
        "failed_files": failed_files
    }


def lambda_handler(event, context):
    event = event or {}

    # Manual backfill mode: scan all QMJ Parquet files in Processed Zone.
    if event.get("mode") == "backfill":
        bucket = event.get("bucket", PROCESSED_BUCKET)
        prefix = event.get("prefix", PROCESSED_PREFIX)
        keys = get_existing_parquet_files(bucket, prefix)

        if not keys:
            return {
                "mode": "backfill",
                "processed_files": 0,
                "loaded_rows": 0,
                "skipped_rows": 0,
                "failed_files": []
            }

        result = process_files(bucket, keys)
        result["mode"] = "backfill"
        result["total_parquet_files_found"] = len(keys)
        return result

    # S3 trigger mode: filter ObjectCreated events in the Processed QMJ prefix.
    keys_by_bucket = {}

    for record in event.get("Records", []):
        if record.get("eventSource") != "aws:s3":
            continue

        if not record.get("eventName", "").startswith("ObjectCreated:"):
            continue

        bucket = record["s3"]["bucket"]["name"]
        key = urllib.parse.unquote_plus(record["s3"]["object"]["key"])

        if not key.startswith(PROCESSED_PREFIX) or not key.endswith(".parquet"):
            continue

        keys_by_bucket.setdefault(bucket, []).append(key)

    total_processed_files = 0
    total_loaded_rows = 0
    total_skipped_rows = 0
    failed_files = []

    for bucket, keys in keys_by_bucket.items():
        result = process_files(bucket, keys)
        total_processed_files += result["processed_files"]
        total_loaded_rows += result["loaded_rows"]
        total_skipped_rows += result["skipped_rows"]
        failed_files.extend(result["failed_files"])

    response = {
        "mode": "s3_trigger",
        "processed_files": total_processed_files,
        "loaded_rows": total_loaded_rows,
        "skipped_rows": total_skipped_rows,
        "failed_files": failed_files
    }

    if failed_files:
        raise Exception(json.dumps(response))

    return response
````
