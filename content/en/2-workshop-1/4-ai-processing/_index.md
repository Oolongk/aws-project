At the Transformation layer, raw JSON data from S3 will be analyzed, summarized, and automatically sentiment-labeled. The system uses an Event-Driven architecture with an S3 Trigger to activate the `Transformer_Lambda` function. Natural language processing (NLP) is handled by the **Claude 4.5 Haiku** model via **Amazon Bedrock**.

### Goals of this stage:
1. Configure LLM model access on Amazon Bedrock.
2. Set up the data processing Lambda function and S3 Event Trigger.
3. Deploy source code optimized for token cost and JSON error handling (Data Chunking & Error Handling).

### Step 3: Deploy Transformation Source Code

The code below solves 3 major challenges when working with LLMs in a Production environment:
* **Cost Optimization:** Limit output tokens by asking AI to return a summary (10-15 words) instead of a full translation.
* **Data Chunking:** Split the comment list into small batches (15 items/batch) to avoid exceeding Bedrock's `max_tokens`.
* **JSON Parsing Fallback:** Build a 3-layer filter to safely catch and handle truncated JSON strings generated by AI errors.

Deploy the following code into the `Transformer_Lambda` function and click **Deploy**:

```python
import json
import boto3
import urllib.parse
import time

s3 = boto3.client('s3')
# Explicitly specify us-east-1 Region to use Inference Profile
bedrock = boto3.client('bedrock-runtime', region_name='us-east-1') 

def analyze_sentiment_batch(comments):
    prompt = f"""You are a data analysis expert.
    Read the following YouTube comments and perform:
    1. Identify the original language.
    2. Evaluate sentiment (POSITIVE, NEGATIVE, NEUTRAL).
    3. Summarize the main point in English (maximum 15 words). Ignore slang and symbols.

    RETURN ONLY 1 VALID JSON ARRAY. NO MARKDOWN.
    Required structure: "id", "language", "vietnamese_summary", "sentiment".
    
    Input data:
    {json.dumps(comments, ensure_ascii=False)}
    """
    
    body = json.dumps({
        "anthropic_version": "bedrock-2023-05-31",
        "max_tokens": 4000, 
        "messages": [{"role": "user", "content": prompt}],
        "temperature": 0 
    })

    try:
        response = bedrock.invoke_model(
            modelId='us.anthropic.claude-3-5-haiku-20241022-v1:0', 
            body=body
        )
        response_body = json.loads(response.get('body').read())
        ai_response_text = response_body['content'][0]['text'].strip()
        
        # Safe JSON string parsing logic
        start_idx = ai_response_text.find('[')
        end_idx = ai_response_text.rfind(']')
        
        if start_idx != -1 and end_idx != -1 and end_idx > start_idx:
            clean_json = ai_response_text[start_idx:end_idx + 1]
            sentiment_results = json.loads(clean_json)
            
            return {
                item['id']: {
                    'sentiment': item.get('sentiment', 'UNKNOWN'),
                    'language': item.get('language', 'Unknown'),
                    'vietnamese_text': item.get('vietnamese_summary', 'No summary')
                } for item in sentiment_results
            }
        else:
            print(f"[ERROR] Invalid JSON response from LLM: {ai_response_text}")
            return {}
    except Exception as e:
        print(f"[ERROR] Bedrock Invocation Failed: {e}")
        return {}

def chunk_list(lst, chunk_size):
    for i in range(0, len(lst), chunk_size):
        yield lst[i:i + chunk_size]

def lambda_handler(event, context):
    for record in event['Records']:
        bucket_name = record['s3']['bucket']['name']
        file_key = urllib.parse.unquote_plus(record['s3']['object']['key'])
        
        response = s3.get_object(Bucket=bucket_name, Key=file_key)
        raw_data = json.loads(response['Body'].read().decode('utf-8'))
        
        all_comments = [{"id": item['id'], "text": item['textOriginal']} for item in raw_data.get('items', [])]
        if not all_comments:
            continue
            
        final_analysis_mapping = {}
        batch_size = 15 # Batch limit
        chunks = list(chunk_list(all_comments, batch_size))
        
        for index, chunk in enumerate(chunks):
            mapping = analyze_sentiment_batch(chunk)
            final_analysis_mapping.update(mapping)
            if index < len(chunks) - 1:
                time.sleep(1) # Prevent Rate Limiting errors from Bedrock API
        
        processed_items = []
        for item in raw_data.get('items', []):
            analysis = final_analysis_mapping.get(item['id'], {})
            item['sentiment'] = analysis.get('sentiment', 'UNKNOWN')
            item['language'] = analysis.get('language', 'Unknown')
            item['vietnamese_text'] = analysis.get('vietnamese_text', 'Translation error')
            processed_items.append(item)
            
        raw_data['items'] = processed_items
        
        # Move data to processed_data partition
        path_parts = file_key.split('/') 
        path_parts[0] = 'processed_data' 
        new_file_key = '/'.join(path_parts) 
        
        s3.put_object(Bucket=bucket_name, Key=new_file_key, Body=json.dumps(raw_data, ensure_ascii=False))
        print(f"[SUCCESS] Processed data saved to: {new_file_key}")

    return {'statusCode': 200, 'body': "Transformation Completed."}
```
